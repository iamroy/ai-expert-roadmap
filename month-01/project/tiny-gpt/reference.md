```python
"""Small CPU teaching model; no remote data or pretrained weights required."""
import argparse
import math
import time
from dataclasses import asdict, dataclass
from pathlib import Path

import torch
import torch.nn as nn
import torch.nn.functional as F


@dataclass(frozen=True)
class Config:
    vocab_size: int = 256
    context: int = 64
    width: int = 64
    heads: int = 4
    layers: int = 2
    dropout: float = 0.1

    def __post_init__(self):
        sizes = (self.vocab_size, self.context, self.width, self.heads, self.layers)
        if any(isinstance(x, bool) or not isinstance(x, int) or x <= 0 for x in sizes):
            raise ValueError("model sizes must be positive integers")
        if self.width % self.heads:
            raise ValueError("width must be divisible by heads")
        if not 0 <= self.dropout < 1:
            raise ValueError("dropout must be in [0, 1)")


def encode(text):
    return list(text.encode("utf-8"))


def decode(ids):
    # Arbitrary generated bytes may be invalid UTF-8; replacement is deliberate.
    return bytes(ids).decode("utf-8", errors="replace")


def make_streams():
    subjects = ["the cat", "a bird", "the robot", "a student", "the dog", "a child"]
    verbs = ["sees", "finds", "likes", "moves", "counts"]
    objects = ["a red ball", "the blue box", "a small toy", "the green cube", "a book"]
    documents = [f"{s} {v} {o}.\n" for s in subjects for v in verbs for o in objects]
    order = torch.randperm(len(documents), generator=torch.Generator().manual_seed(123)).tolist()
    cut = int(len(order) * 0.8)
    train_docs = [documents[i] for i in order[:cut]]
    val_docs = [documents[i] for i in order[cut:]]
    assert set(train_docs).isdisjoint(val_docs)
    return tuple(
        torch.tensor(encode("".join(docs)), dtype=torch.long)
        for docs in (train_docs, val_docs)
    )


def batch(stream, batch_size, context, generator):
    if stream.ndim != 1 or stream.dtype != torch.long or stream.device.type != "cpu":
        raise ValueError("expected a 1D CPU long token stream")
    if batch_size <= 0 or context <= 0 or len(stream) <= context:
        raise ValueError("positive sizes and at least context+1 stream tokens required")
    starts = torch.randint(len(stream) - context, (batch_size,), generator=generator)
    x = torch.stack([stream[i:i + context] for i in starts.tolist()])
    y = torch.stack([stream[i + 1:i + context + 1] for i in starts.tolist()])
    return x, y


class Attention(nn.Module):
    def __init__(self, cfg):
        super().__init__()
        self.heads = cfg.heads
        self.head_dim = cfg.width // cfg.heads
        self.qkv = nn.Linear(cfg.width, 3 * cfg.width)
        self.out = nn.Linear(cfg.width, cfg.width)
        self.attn_dropout = nn.Dropout(cfg.dropout)
        self.out_dropout = nn.Dropout(cfg.dropout)
        self.register_buffer(
            "allowed", torch.ones(cfg.context, cfg.context, dtype=torch.bool).tril()
        )

    def forward(self, x, capture=False):
        b, t, d = x.shape
        q, k, v = [
            part.reshape(b, t, self.heads, self.head_dim).transpose(1, 2)
            for part in self.qkv(x).chunk(3, dim=-1)
        ]
        scores = q @ k.transpose(-2, -1) / math.sqrt(self.head_dim)
        weights = scores.masked_fill(~self.allowed[:t, :t], float("-inf")).softmax(-1)
        values = self.attn_dropout(weights) @ v
        merged = values.transpose(1, 2).contiguous().view(b, t, d)
        debug = None
        if capture:
            debug = {
                "q": q.detach(), "k": k.detach(), "v": v.detach(),
                "attention": weights.detach(),
            }
        return self.out_dropout(self.out(merged)), debug


class Block(nn.Module):
    def __init__(self, cfg):
        super().__init__()
        self.ln1 = nn.LayerNorm(cfg.width)
        self.ln2 = nn.LayerNorm(cfg.width)
        self.attention = Attention(cfg)
        self.ffn = nn.Sequential(
            nn.Linear(cfg.width, 4 * cfg.width), nn.GELU(),
            nn.Linear(4 * cfg.width, cfg.width), nn.Dropout(cfg.dropout),
        )

    def forward(self, x, capture=False):
        update, debug = self.attention(self.ln1(x), capture)
        x = x + update
        x = x + self.ffn(self.ln2(x))
        if capture:
            debug["block_output"] = x.detach()
        return x, debug


class TinyGPT(nn.Module):
    def __init__(self, cfg):
        super().__init__()
        self.cfg = cfg
        self.tokens = nn.Embedding(cfg.vocab_size, cfg.width)
        self.positions = nn.Embedding(cfg.context, cfg.width)
        self.blocks = nn.ModuleList([Block(cfg) for _ in range(cfg.layers)])
        self.norm = nn.LayerNorm(cfg.width)
        self.head = nn.Linear(cfg.width, cfg.vocab_size, bias=False)
        self.apply(self._initialize)

    @staticmethod
    def _initialize(module):
        if isinstance(module, (nn.Linear, nn.Embedding)):
            nn.init.normal_(module.weight, std=0.02)
        if isinstance(module, nn.Linear) and module.bias is not None:
            nn.init.zeros_(module.bias)

    def forward(self, ids, capture=False):
        if ids.ndim != 2 or ids.dtype != torch.long or ids.size(0) == 0:
            raise ValueError("expected nonempty [batch, time] long token IDs")
        if not 1 <= ids.size(1) <= self.cfg.context:
            raise ValueError("input length outside configured context")
        if (ids < 0).any() or (ids >= self.cfg.vocab_size).any():
            raise ValueError("token ID outside vocabulary")
        positions = torch.arange(ids.size(1), device=ids.device)
        x = self.tokens(ids) + self.positions(positions)
        traces = []
        for block in self.blocks:
            x, trace = block(x, capture)
            if capture:
                traces.append(trace)
        return self.head(self.norm(x)), traces


def token_loss(logits, targets):
    return F.cross_entropy(
        logits.reshape(-1, logits.size(-1)), targets.reshape(-1)
    )


def sampling_probs(logits, temperature=1.0, top_k=None, top_p=1.0):
    if logits.ndim != 1 or logits.numel() == 0 or not torch.isfinite(logits).all():
        raise ValueError("expected a nonempty finite 1D logit vector")
    if not math.isfinite(temperature) or temperature <= 0:
        raise ValueError("temperature must be finite and positive")
    if not 0 < top_p <= 1:
        raise ValueError("top_p must be in (0, 1]")
    scores = logits.float() / temperature
    if not torch.isfinite(scores).all():
        raise ValueError("temperature caused logit overflow")
    if top_k is not None:
        if (
            isinstance(top_k, bool) or not isinstance(top_k, int)
            or not 1 <= top_k <= scores.numel()
        ):
            raise ValueError("top_k must be an integer in [1, vocabulary size]")
        keep = torch.topk(scores, top_k).indices
        filtered = torch.full_like(scores, float("-inf"))
        filtered[keep] = scores[keep]
        scores = filtered
    ordered, indices = torch.sort(scores, descending=True)
    probabilities = ordered.softmax(dim=-1)
    mass_before = probabilities.cumsum(dim=-1) - probabilities
    if top_p < 1:
        ordered = ordered.masked_fill(mass_before >= top_p, float("-inf"))
    ordered_probabilities = ordered.softmax(dim=-1)
    result = torch.zeros_like(ordered_probabilities)
    result.scatter_(0, indices, ordered_probabilities)
    return result


@torch.no_grad()
def generate(model, prompt, count, mode="greedy", temperature=1.0,
             top_k=20, top_p=0.9, seed=7):
    if not prompt or not isinstance(count, int) or count < 0:
        raise ValueError("nonempty prompt and nonnegative integer count required")
    if mode not in {"greedy", "temperature", "top_k", "top_p"}:
        raise ValueError("unknown generation mode")
    if model.cfg.vocab_size != 256 or next(model.parameters()).device.type != "cpu":
        raise ValueError("the byte generation helper supports a CPU 256-byte model")
    if mode != "greedy":
        sampling_probs(
            torch.zeros(256), temperature,
            top_k if mode == "top_k" else None,
            top_p if mode == "top_p" else 1.0,
        )
    ids = torch.tensor([encode(prompt)], dtype=torch.long)
    generator = torch.Generator().manual_seed(seed)
    was_training = model.training
    model.eval()
    try:
        for _ in range(count):
            logits, _ = model(ids[:, -model.cfg.context:])
            last = logits[0, -1]
            if mode == "greedy":
                next_id = last.argmax().view(1)
            else:
                probabilities = sampling_probs(
                    last, temperature,
                    top_k if mode == "top_k" else None,
                    top_p if mode == "top_p" else 1.0,
                )
                next_id = torch.multinomial(probabilities, 1, generator=generator)
            ids = torch.cat((ids, next_id.view(1, 1)), dim=1)
        return {
            "token_ids": ids[0].tolist(), "text": decode(ids[0].tolist()),
            "generated_tokens": count, "finish_reason": "length",
        }
    finally:
        model.train(was_training)


@torch.no_grad()
def evaluate(model, stream, batch_size=8, batches=8):
    if batches <= 0:
        raise ValueError("evaluation needs at least one batch")
    generator = torch.Generator().manual_seed(999)
    was_training = model.training
    model.eval()
    total_loss = 0.0
    count = 0
    try:
        for _ in range(batches):
            x, y = batch(stream, batch_size, model.cfg.context, generator)
            logits, _ = model(x)
            total_loss += F.cross_entropy(
                logits.reshape(-1, model.cfg.vocab_size), y.reshape(-1),
                reduction="sum",
            ).item()
            count += y.numel()
        return total_loss / count
    finally:
        model.train(was_training)


def save_model(model, path):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    torch.save({
        "config": asdict(model.cfg), "state": model.state_dict(),
        "tokenizer": "utf8-bytes-v1",
    }, path)


def load_model(path):
    checkpoint = torch.load(path, map_location="cpu", weights_only=True)
    if checkpoint["tokenizer"] != "utf8-bytes-v1":
        raise ValueError("unsupported tokenizer")
    model = TinyGPT(Config(**checkpoint["config"]))
    model.load_state_dict(checkpoint["state"])
    return model.eval()


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--steps", type=int, default=200)
    parser.add_argument("--seed", type=int, default=7)
    parser.add_argument("--checkpoint", default="checkpoints/tiny-gpt.pt")
    args = parser.parse_args()
    if args.steps < 1:
        parser.error("--steps must be positive")

    torch.set_num_threads(1)
    torch.manual_seed(args.seed)
    train_stream, validation_stream = make_streams()
    config = Config()
    model = TinyGPT(config)
    optimizer = torch.optim.AdamW(model.parameters(), lr=3e-3)
    generator = torch.Generator().manual_seed(args.seed)
    print("config", asdict(config))
    print("parameters", sum(parameter.numel() for parameter in model.parameters()))
    print("initial train/validation", evaluate(model, train_stream),
          evaluate(model, validation_stream))

    started = time.perf_counter()
    for _ in range(args.steps):
        model.train()
        x, y = batch(train_stream, 8, config.context, generator)
        optimizer.zero_grad(set_to_none=True)
        logits, _ = model(x)
        loss = token_loss(logits, y)
        if not torch.isfinite(loss):
            raise RuntimeError("nonfinite training loss")
        loss.backward()
        nn.utils.clip_grad_norm_(model.parameters(), 1.0)
        optimizer.step()
    elapsed = time.perf_counter() - started
    processed = args.steps * 8 * config.context
    print("final train/validation", evaluate(model, train_stream),
          evaluate(model, validation_stream))
    print("training tokens", processed)
    print("training seconds/tokens_per_second", elapsed, processed / elapsed)

    model.eval()
    with torch.no_grad():
        logits, traces = model(x[:1], capture=True)
    print("logits", tuple(logits.shape))
    print("first block", {name: tuple(value.shape) for name, value in traces[0].items()})
    save_model(model, args.checkpoint)
    restored = load_model(args.checkpoint)
    with torch.no_grad():
        torch.testing.assert_close(model(x[:1])[0], restored(x[:1])[0])
    print("checkpoint reload: passed")
    for mode in ("greedy", "temperature", "top_k", "top_p"):
        result = generate(restored, "the cat ", 80, mode=mode, seed=args.seed)
        print(mode, result["finish_reason"], repr(result["text"]))


if __name__ == "__main__":
    main()
```

```python
import tempfile
import unittest
from pathlib import Path

import torch
import torch.nn.functional as F

from tiny_gpt import (
    Config, TinyGPT, batch, decode, encode, evaluate, generate, load_model,
    make_streams, sampling_probs, save_model, token_loss,
)


class TinyGPTTests(unittest.TestCase):
    def setUp(self):
        torch.set_num_threads(1)
        torch.manual_seed(11)
        self.config = Config(context=8, width=16, heads=4, layers=1, dropout=0.0)
        self.model = TinyGPT(self.config)
        self.ids = torch.tensor([encode("the cat ")], dtype=torch.long)

    def test_tokenizer_and_shift(self):
        for text in ("", "hello", "café", "東京", "🙂", "a\n b"):
            self.assertEqual(decode(encode(text)), text)
        stream = torch.arange(5, dtype=torch.long)
        x, y = batch(stream, 1, 4, torch.Generator().manual_seed(2))
        self.assertEqual(x.tolist(), [[0, 1, 2, 3]])
        self.assertEqual(y.tolist(), [[1, 2, 3, 4]])

    def test_shapes_attention_and_causality(self):
        self.model.eval()
        logits, traces = self.model(self.ids, capture=True)
        self.assertEqual(tuple(logits.shape), (1, 8, 256))
        for name in ("q", "k", "v"):
            self.assertEqual(tuple(traces[0][name].shape), (1, 4, 8, 4))
        weights = traces[0]["attention"]
        torch.testing.assert_close(weights.sum(-1), torch.ones(1, 4, 8))
        self.assertEqual(weights.triu(1).count_nonzero().item(), 0)
        expected = F.scaled_dot_product_attention(
            traces[0]["q"], traces[0]["k"], traces[0]["v"],
            is_causal=True, dropout_p=0.0,
        )
        torch.testing.assert_close(weights @ traces[0]["v"], expected)
        changed = self.ids.clone()
        changed[:, 4:] = torch.tensor([5, 6, 7, 8])
        with torch.no_grad():
            first, _ = self.model(self.ids)
            second, _ = self.model(changed)
        torch.testing.assert_close(first[:, :4], second[:, :4])

    def test_loss_gradient_and_overfit(self):
        probabilities = torch.tensor([[0.2, 0.5, 0.3]])
        logits = probabilities.log().view(1, 1, 3).requires_grad_()
        loss = token_loss(logits, torch.tensor([[1]]))
        loss.backward()
        torch.testing.assert_close(loss, torch.tensor(2.0).log())
        torch.testing.assert_close(logits.grad.flatten(), torch.tensor([0.2, -0.5, 0.3]))

        targets = torch.tensor([encode("he cat .")], dtype=torch.long)
        optimizer = torch.optim.AdamW(self.model.parameters(), lr=0.015)
        initial = token_loss(self.model(self.ids)[0], targets).item()
        for _ in range(60):
            optimizer.zero_grad(set_to_none=True)
            loss = token_loss(self.model(self.ids)[0], targets)
            loss.backward()
            optimizer.step()
        self.assertLess(token_loss(self.model(self.ids)[0], targets).item(), initial * 0.4)

    def test_sampler_and_generation(self):
        logits = torch.tensor([0.55, 0.25, 0.15, 0.05]).log()
        torch.testing.assert_close(
            sampling_probs(logits, top_k=2), torch.tensor([0.6875, 0.3125, 0., 0.])
        )
        torch.testing.assert_close(
            sampling_probs(logits, top_p=0.9),
            torch.tensor([0.55, 0.25, 0.15, 0.]) / 0.95,
        )
        torch.testing.assert_close(
            sampling_probs(logits, top_p=0.2), torch.tensor([1., 0., 0., 0.])
        )
        for invalid in ({"temperature": 0}, {"top_p": 0}, {"top_k": 0}):
            with self.assertRaises(ValueError):
                sampling_probs(torch.zeros(4), **invalid)
        for mode in ("greedy", "temperature", "top_k", "top_p"):
            first = generate(self.model, "the cat ", 5, mode=mode, seed=9)
            second = generate(self.model, "the cat ", 5, mode=mode, seed=9)
            self.assertEqual(first, second)

    def test_validation_and_checkpoint(self):
        _, validation_stream = make_streams()
        self.model.train()
        self.assertEqual(
            evaluate(self.model, validation_stream, batches=2),
            evaluate(self.model, validation_stream, batches=2),
        )
        self.assertTrue(self.model.training)
        self.model.eval()
        with tempfile.TemporaryDirectory() as folder:
            path = Path(folder) / "model.pt"
            save_model(self.model, path)
            restored = load_model(path)
            with torch.no_grad():
                torch.testing.assert_close(self.model(self.ids)[0], restored(self.ids)[0])


if __name__ == "__main__":
    unittest.main()
```

## Verification record

The full reference and an expanded 13-test audit were run with Python 3.12.10 and PyTorch 2.14.0 on CPU. The audit covered tokenizer behavior, shifted windows, Q/K/V and logit shapes, explicit attention against PyTorch scaled dot-product attention, causal invariance, manual cross-entropy gradients, fixed-batch overfitting, sampler filters and invalid settings, deterministic generation, validation-mode restoration, invalid model inputs, and checkpoint equivalence.

This reference uses deliberately narrow synthetic data. Its loss decrease demonstrates that the mechanics work; it does not measure general language ability.
